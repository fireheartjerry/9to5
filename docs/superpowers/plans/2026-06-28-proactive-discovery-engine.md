> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a proactive discovery phase to 9TO5 that compiles a codebase digest, runs 5 parallel innovation lenses via Ollama, synthesizes an impact-ranked report, and feeds it to the architect — so vague goals like "improve everything" produce bold, coherent PRs instead of random chores.

**Architecture:** New `agents/discovery/` module (models → digest → lenses → synthesis → facade). The facade is called from `run_phase_mixin._freshRunPhase()` before the architect, storing the report in `state.meta["discovery_report"]`. The architect prompt is augmented with the report when present.

**Tech Stack:** Python stdlib (`subprocess`, `concurrent.futures`, `json`, `pathlib`), Ollama (same pattern as `agents/elaborator.py`), existing `llm_io.responseText`, `config`, `memory.experience`.

```python
# agents/discovery/models.py
"""Finding and DiscoveryReport dataclasses for the discovery engine.

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  Finding  -- One discovered improvement opportunity from a lens agent.
  DiscoveryReport  -- Ranked synthesis of all lens findings.
DEPENDS-ON: (stdlib only)
SIDE-EFFECTS: unannotated
RUN/TEST: python -m pytest tests/test_discovery_models.py
"""
from __future__ import annotations

from dataclasses import dataclass, field
from typing import List, Optional


@dataclass
class Finding:
    title: str
    type: str          # new_capability | reimagining | health | memory | ux
    impact: int        # 1-5
    specificity: int   # 1-5
    innovation: int    # 1-5
    rationale: str
    affected_files: List[str]
    source_lens: str
    rank: Optional[int] = None
    composes_with: List[int] = field(default_factory=list)

    def to_dict(self) -> dict:
        return {
            "rank": self.rank,
            "title": self.title,
            "type": self.type,
            "impact": self.impact,
            "specificity": self.specificity,
            "innovation": self.innovation,
            "rationale": self.rationale,
            "affected_files": self.affected_files,
            "source_lens": self.source_lens,
            "composes_with": self.composes_with,
        }


@dataclass
class DiscoveryReport:
    theme_suggestion: str
    findings: List[Finding]

    @property
    def innovation_ratio(self) -> float:
        if not self.findings:
            return 0.0
        innovative = sum(1 for f in self.findings if f.innovation >= 3)
        return innovative / len(self.findings)

    def to_dict(self) -> dict:
        return {
            "theme_suggestion": self.theme_suggestion,
            "innovation_ratio": round(self.innovation_ratio, 2),
            "findings": [f.to_dict() for f in self.findings],
        }
```

```python
# agents/discovery/digest.py
"""Phase 0: compile the shared codebase digest for all lens agents.

Pure tool calls — subprocess, file reads, grep. No LLM. Compiled once,
shared by all 5 lens agents. Errors in any sub-collector are caught and
return empty values so a partial environment never crashes a run.

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  compile_digest(work_dir: str) -> dict  -- Compile and return the full codebase digest.
DEPENDS-ON: config, memory.experience
SIDE-EFFECTS: unannotated
RUN/TEST: python -m pytest tests/test_discovery_digest.py
"""
from __future__ import annotations

import json
import subprocess
from pathlib import Path

import config

LOG = config.getLogger("discovery.digest")

_TIMEOUT = 15  # seconds per subprocess call


def _run(cmd: list, cwd: str = None) -> str:
    """Run a subprocess and return stdout. Returns '' on any error."""
    try:
        r = subprocess.run(cmd, capture_output=True, text=True,
                           timeout=_TIMEOUT, cwd=cwd)
        return (r.stdout or "").strip()
    except Exception as e:  # noqa: BLE001
        LOG.debug("digest subprocess failed %s: %s", cmd[0], e)
        return ""


def _grep(pattern: str, path: str, extra_args: list = None) -> str:
    args = ["grep", "-r", "-n", "--include=*.py", pattern, path]
    if extra_args:
        args += extra_args
    return _run(args)


def _read_file(path: str) -> str:
    try:
        return Path(path).read_text(encoding="utf-8", errors="replace")
    except Exception:  # noqa: BLE001
        return ""


# ── Repo structure ────────────────────────────────────────────────────────────

def _read_repo_map(work_dir: str) -> dict:
    raw = _read_file(str(Path(work_dir) / ".repo-map.json"))
    try:
        return json.loads(raw) if raw else {}
    except Exception:  # noqa: BLE001
        return {}


def _read_architecture(work_dir: str) -> str:
    return _read_file(str(Path(work_dir) / "docs" / "ARCHITECTURE.md"))[:8000]


def _extract_module_signatures(work_dir: str) -> str:
    """Top-level EXPORTS sections from all @AI-MAP headers."""
    return _grep("EXPORTS:", work_dir)[:3000]


def _build_dependency_graph(work_dir: str) -> str:
    """DEPENDS-ON lines from all @AI-MAP headers."""
    return _grep("DEPENDS-ON:", work_dir)[:3000]


def _find_dead_code(work_dir: str) -> str:
    """__all__ exports and public symbols — cross-ref is done by lenses."""
    return _grep(r"^def [a-z]", work_dir)[:2000]


# ── Static signals ────────────────────────────────────────────────────────────

def _collect_static_signals(work_dir: str) -> dict:
    size_out = _run(["python", "tools/size_audit.py"], cwd=work_dir)
    todos = _grep(r"TODO\|FIXME\|HACK\|XXX\|DEPRECATED", work_dir)
    # Limit to most recent 30 by taking last 30 lines (grep already sorts by file)
    todo_lines = todos.splitlines()
    todos_trimmed = "\n".join(todo_lines[:30])

    docs_drift = _grep(r"@AI-MAP", work_dir)[:1000]
    test_gaps_raw = _grep(r"^def [a-z]", str(Path(work_dir) / "tests"))[:2000]

    isolation = _grep(
        r"import discord\|import ollama\|import requests",
        str(Path(work_dir) / "tests"),
    )[:1000]

    deep_nesting = _run(
        ["python", "-m", "radon", "cc", "-n", "C", "-a", work_dir],
        cwd=work_dir,
    )[:1000]

    long_fns = _grep(r"^def ", work_dir)[:1000]

    security = _grep(r"eval(\|exec(\|os\.system(\|password\s*=\s*[\"']", work_dir)[:500]

    return {
        "size_violations": size_out[:2000],
        "todos": todos_trimmed,
        "docs_drift": docs_drift,
        "test_gaps": test_gaps_raw,
        "test_isolation_violations": isolation,
        "deep_nesting": deep_nesting,
        "long_functions": long_fns,
        "security_signals": security,
    }


def _collect_quality_signals(work_dir: str) -> dict:
    coverage = _read_file(str(Path(work_dir) / "logs" / "coverage.txt"))[:1000]
    slow_tests = _read_file(str(Path(work_dir) / "logs" / "slow-test-report.md"))[:1000]
    workflow_out = _run(["python", "validate_workflows.py"], cwd=work_dir)[:1000]
    return {
        "coverage": coverage,
        "slow_tests": slow_tests,
        "workflow_registry": workflow_out,
        "flaky_history": "",  # populated from institutional_memory
    }


# ── Git + GitHub signals ──────────────────────────────────────────────────────

def _collect_git_signals(work_dir: str) -> dict:
    churn = _run(
        ["git", "log", "--name-only", "--format=", "--diff-filter=M", "-n", "30"],
        cwd=work_dir,
    )[:2000]
    bug_hotspots = _run(
        ["git", "log", "--name-only", "--format=", "--grep=fix", "-n", "30"],
        cwd=work_dir,
    )[:1000]
    recent = _run(
        ["git", "log", "--name-only", "--format=", "-n", "10"],
        cwd=work_dir,
    )[:1000]
    stale = _run(
        ["git", "branch", "-a", "--sort=-committerdate",
         "--format=%(refname:short) %(committerdate:relative)"],
        cwd=work_dir,
    )[:500]
    return {
        "churn": churn,
        "bug_fix_hotspots": bug_hotspots,
        "recent_touches": recent,
        "stale_branches": stale,
    }


def _collect_github_signals(work_dir: str) -> dict:
    issues = _run(["gh", "issue", "list", "--limit", "20",
                   "--json", "title,labels,createdAt"], cwd=work_dir)[:2000]
    prs = _run(["gh", "pr", "list", "--limit", "10",
                "--json", "title,createdAt,url"], cwd=work_dir)[:1000]
    ci = _run(["gh", "run", "list", "--limit", "20",
               "--json", "name,conclusion,createdAt"], cwd=work_dir)[:1500]
    return {
        "open_issues": issues,
        "stale_prs": prs,
        "ci_failure_patterns": ci,
        "recent_pr_review_comments": "",
    }


# ── Institutional memory ──────────────────────────────────────────────────────

def _collect_institutional_memory() -> dict:
    try:
        from memory import experience
        lessons = experience.recent_reflections(limit=15)
    except Exception:  # noqa: BLE001
        lessons = []
    try:
        from memory import experience
        failures = experience.recent_failures(limit=10)
    except Exception:  # noqa: BLE001
        failures = []
    try:
        from learning import postmortem
        open_pm = postmortem.open_guidance()
    except Exception:  # noqa: BLE001
        open_pm = []
    return {
        "recent_lessons": lessons,
        "recent_failures": failures,
        "open_postmortems": open_pm,
        "policy_hints": "",
        "knowledge_graph": "",
        "discord_feedback": "",
    }


# ── Innovation seeds ──────────────────────────────────────────────────────────

def _collect_innovation_seeds(work_dir: str) -> dict:
    parked = _grep("PARKED", str(Path(work_dir) / "docs" / "plans"))[:1000]
    arch = _read_architecture(work_dir)
    underused = _grep(r"skills_to_invoke", work_dir)[:500]
    prompts = _grep(r"_SYSTEM\s*=", work_dir)[:1000]
    return {
        "vision_vs_reality": arch[:3000],
        "rejected_ideas": "",
        "parked_plans": parked,
        "underutilized_capabilities": underused,
        "prompt_quality_signals": prompts,
    }


def _collect_leverage_signals(work_dir: str) -> dict:
    slow_log = _read_file(str(Path(work_dir) / "logs" / "slow-test-report.md"))[:500]
    repeats = _grep(r"def _", work_dir)[:1000]
    return {
        "slowest_phases": slow_log,
        "highest_token_cost": "",
        "most_retried": "",
        "abstraction_gaps": repeats,
    }


def _collect_skills_hooks(work_dir: str) -> str:
    hooks = _read_file(str(Path(work_dir) / ".claude" / "settings.json"))[:500]
    skills = _run(["ls", str(Path(work_dir) / ".claude" / "skills")])[:300]
    return (hooks + "\n" + skills)[:800]


def _collect_config_signals(work_dir: str) -> str:
    return _grep(r"^[A-Z_]+ =", str(Path(work_dir) / "config.py"))[:1000]


# ── Public facade ─────────────────────────────────────────────────────────────

def compile_digest(work_dir: str) -> dict:
    """Compile and return the full codebase digest. Never raises."""
    LOG.info("Compiling discovery digest for %s", work_dir)
    try:
        return {
            "repo_map": _read_repo_map(work_dir),
            "architecture": _read_architecture(work_dir),
            "module_signatures": _extract_module_signatures(work_dir),
            "dependency_graph": _build_dependency_graph(work_dir),
            "dead_code_signals": _find_dead_code(work_dir),
            "static_signals": _collect_static_signals(work_dir),
            "quality": _collect_quality_signals(work_dir),
            "git_signals": _collect_git_signals(work_dir),
            "github": _collect_github_signals(work_dir),
            "institutional_memory": _collect_institutional_memory(),
            "innovation_seeds": _collect_innovation_seeds(work_dir),
            "leverage_signals": _collect_leverage_signals(work_dir),
            "skills_hooks": _collect_skills_hooks(work_dir),
            "config_signals": _collect_config_signals(work_dir),
        }
    except Exception as e:  # noqa: BLE001
        LOG.warning("compile_digest failed (%s); returning empty digest", e)
        return {k: {} for k in (
            "repo_map", "architecture", "module_signatures", "dependency_graph",
            "dead_code_signals", "static_signals", "quality", "git_signals",
            "github", "institutional_memory", "innovation_seeds",
            "leverage_signals", "skills_hooks", "config_signals",
        )}
```

```python
# agents/discovery/lenses.py
"""Phase 1: parallel lens panel — 5 agents run concurrently on the digest.

Each lens produces 0-3 Finding dicts via Ollama. Lenses run in a thread pool;
individual failures are caught so a dead lens never blocks the others.

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  run_lens_panel(digest: dict) -> list[Finding]  -- Run all 5 lenses in parallel.
DEPENDS-ON: agents.discovery.models, config, llm_io
SIDE-EFFECTS: unannotated
RUN/TEST: python -m pytest tests/test_discovery_lenses.py
"""
from __future__ import annotations

import json
import time
from concurrent.futures import ThreadPoolExecutor, as_completed
from typing import List

import config
import llm_io
from agents.discovery.models import Finding

LOG = config.getLogger("discovery.lenses")

_OLLAMA = None
_VALID_TYPES = {"new_capability", "reimagining", "health", "memory", "ux"}
_MAX_FINDINGS_PER_LENS = 3


def _getOllama():
    global _OLLAMA
    if _OLLAMA is None:
        import ollama
        _OLLAMA = ollama.Client(host=config.OLLAMA_HOST, timeout=config.OLLAMA_TIMEOUT_SECONDS)
    return _OLLAMA


def _brain_call(system: str, user: str, retries: int = 2) -> list:
    """Call Ollama and parse the findings JSON list. Returns [] on any failure."""
    delay = 2.0
    for attempt in range(1, retries + 1):
        try:
            resp = _getOllama().chat(
                model=config.OLLAMA_MODEL,
                messages=[{"role": "system", "content": system},
                          {"role": "user", "content": user}],
                format="json",
                options={"num_ctx": config.OLLAMA_NUM_CTX},
                keep_alive=config.OLLAMA_KEEP_ALIVE,
            )
            text = llm_io.responseText(resp).strip()
            parsed = json.loads(text)
            if isinstance(parsed, dict):
                return parsed.get("findings", [])
            return []
        except json.JSONDecodeError:
            LOG.debug("lens brain returned non-JSON (attempt %d)", attempt)
            return []
        except Exception as e:  # noqa: BLE001
            LOG.warning("lens brain call failed (%d/%d): %s", attempt, retries, e)
            if attempt < retries:
                time.sleep(delay)
                delay *= 2
    return []


def _to_finding(raw: dict, source_lens: str) -> Finding | None:
    """Validate and coerce a raw dict into a Finding. Returns None if unfit."""
    try:
        title = str(raw.get("title") or "").strip()
        if not title:
            return None
        ftype = str(raw.get("type") or "health")
        if ftype not in _VALID_TYPES:
            ftype = "health"
        impact = max(1, min(5, int(raw.get("impact") or 3)))
        specificity = max(1, min(5, int(raw.get("specificity") or 3)))
        innovation = max(1, min(5, int(raw.get("innovation") or 1)))
        if specificity < 2:  # too vague to act on
            return None
        return Finding(
            title=title, type=ftype, impact=impact,
            specificity=specificity, innovation=innovation,
            rationale=str(raw.get("rationale") or "")[:500],
            affected_files=[str(f) for f in (raw.get("affected_files") or [])],
            source_lens=source_lens,
        )
    except Exception:  # noqa: BLE001
        return None

def run_lens_panel(digest: dict) -> List[Finding]:
    """Run all lenses in parallel. Returns combined Finding list. Never raises."""
    LOG.info("Running %d discovery lenses in parallel", len(_LENSES))
    all_findings: List[Finding] = []
    with ThreadPoolExecutor(max_workers=len(_LENSES)) as pool:
        futures = {pool.submit(lens, digest): lens.__name__ for lens in _LENSES}
        for future in as_completed(futures):

                LOG.info("Lens %s returned %d findings", name, len(findings))
                all_findings.extend(findings)
            except Exception as e:  # noqa: BLE001
                LOG.warning("Lens %s failed: %s", name, e)
    return all_findings
```

```python
# agents/discovery/synthesis.py
"""Phase 2: synthesize lens findings into a ranked DiscoveryReport.

Deduplicates, scores (impact × specificity × innovation), enforces the
40% innovation floor, and calls Ollama once for a theme suggestion.

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  synthesize(findings: list[Finding]) -> DiscoveryReport  -- Rank and filter findings.
DEPENDS-ON: agents.discovery.models, config, llm_io
SIDE-EFFECTS: unannotated
RUN/TEST: python -m pytest tests/test_discovery_synthesis.py
"""
from __future__ import annotations

import json
import time
from typing import List

import config
import llm_io
from agents.discovery.models import Finding, DiscoveryReport

LOG = config.getLogger("discovery.synthesis")

_OLLAMA = None
_MAX_FINDINGS = 10
_INNOVATION_FLOOR = 0.40


def _getOllama():
    global _OLLAMA
    if _OLLAMA is None:
        import ollama
        _OLLAMA = ollama.Client(host=config.OLLAMA_HOST, timeout=config.OLLAMA_TIMEOUT_SECONDS)
    return _OLLAMA


def _score(f: Finding) -> float:
    return f.impact * f.specificity * f.innovation


def _are_near_duplicates(a: Finding, b: Finding) -> bool:
    """Simple title-based dedup: same first 4 words = duplicate."""
    words_a = a.title.lower().split()[:4]
    words_b = b.title.lower().split()[:4]
    return words_a == words_b


def _dedup(findings: List[Finding]) -> List[Finding]:
    unique: List[Finding] = []
    for f in findings:
        if not any(_are_near_duplicates(f, u) for u in unique):
            unique.append(f)
    return unique


def _enforce_innovation_floor(findings: List[Finding]) -> List[Finding]:
    """Ensure at least 40% of findings score >= 3 on innovation."""
    if not findings:
        return findings
    innovative = [f for f in findings if f.innovation >= 3]
    incremental = [f for f in findings if f.innovation < 3]
    current_ratio = len(innovative) / len(findings)
    if current_ratio >= _INNOVATION_FLOOR:
        return findings
    # Need more innovative: keep all innovative, fill remaining slots from incremental
    # but ensure the ratio is met
    target_innovative = max(1, int(_INNOVATION_FLOOR * len(findings)))
    needed = target_innovative - len(innovative)
    if needed <= 0:
        return findings
    # Re-rank: innovative first, then incremental
    reranked = sorted(innovative, key=_score, reverse=True) + \
               sorted(incremental, key=_score, reverse=True)
    return reranked


def _get_theme(findings: List[Finding]) -> str:
    """Ask Ollama for a one-line theme suggestion. Falls back to deterministic."""
    if not findings:
        return "improve codebase quality"
    titles = [f.title for f in findings[:8]]
    system = (
        "Given these discovered improvement opportunities, suggest ONE short theme "
        "phrase (5-8 words) that captures the most impactful narrative direction. "
        "Output JSON only: {\"theme_suggestion\": \"...\"}"
    )
    user = "Findings:\n" + "\n".join(f"- {t}" for t in titles)
    delay = 1.0
    for attempt in range(1, 3):
        try:
            resp = _getOllama().chat(
                model=config.OLLAMA_MODEL,
                messages=[{"role": "system", "content": system},
                          {"role": "user", "content": user}],
                format="json",
                options={"num_ctx": config.OLLAMA_NUM_CTX},
                keep_alive=config.OLLAMA_KEEP_ALIVE,
            )
            text = llm_io.responseText(resp)
            parsed = json.loads(text)
            theme = str(parsed.get("theme_suggestion") or "").strip()
            if theme:
                return theme
        except Exception as e:  # noqa: BLE001
            LOG.warning("theme synthesis failed (%d/2): %s", attempt, e)
            if attempt < 2:
                time.sleep(delay)
    # Deterministic fallback: use top finding type
    top_type = findings[0].type if findings else "health"
    return f"transform the {top_type.replace('_', ' ')} layer"


def synthesize(findings: List[Finding]) -> DiscoveryReport:
    """Rank and filter findings into a DiscoveryReport. Never raises."""
    LOG.info("Synthesizing %d raw findings", len(findings))
    if not findings:
        return DiscoveryReport(theme_suggestion="improve codebase quality", findings=[])

    # 1. Drop too-vague
    filtered = [f for f in findings if f.specificity >= 2]

    # 2. Deduplicate near-identical titles
    deduped = _dedup(filtered)

    # 3. Score and sort
    scored = sorted(deduped, key=_score, reverse=True)

    # 4. Enforce innovation floor
    balanced = _enforce_innovation_floor(scored)

    # 5. Cap at max findings
    top = balanced[:_MAX_FINDINGS]

    # 6. Assign ranks
    for i, f in enumerate(top):
        f.rank = i + 1

    # 7. Get theme from Ollama
    theme = _get_theme(top)

    LOG.info("Synthesis complete: %d findings, innovation_ratio=%.2f, theme=%r",
             len(top), DiscoveryReport(theme, top).innovation_ratio, theme)
    return DiscoveryReport(theme_suggestion=theme, findings=top)
```

```python
# agents/discovery/__init__.py
"""Proactive Discovery Engine — public facade.

Compile a codebase digest, run parallel innovation lenses, synthesize an
impact-ranked report for the architect to consume.

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  shouldRunFullDiscovery(goal_raw: str) -> bool  -- True when goal is vague/absent.
  run_discovery(work_dir: str, goal_raw: str) -> DiscoveryReport | None  -- Full discovery pipeline.
DEPENDS-ON: agents.discovery.digest, agents.discovery.lenses, agents.discovery.synthesis
SIDE-EFFECTS: unannotated
RUN/TEST: python -m pytest tests/test_discovery_facade.py
"""
from __future__ import annotations

from typing import Optional

import config
from agents.discovery.models import DiscoveryReport

LOG = config.getLogger("discovery")

_VAGUE_PATTERNS = (
    "improve", "fix everything", "do your best", "make it better",
    "optimize", "clean up", "work on the codebase", "improve everything",
    "do something good", "surprise me", "make things better", "enhance",
    "self-improve", "improve self",
)


def shouldRunFullDiscovery(goal_raw: str) -> bool:
    """True when goal is absent or matches a vague pattern."""
    if not goal_raw or not goal_raw.strip():
        return True
    low = goal_raw.lower().strip()
    return any(p in low for p in _VAGUE_PATTERNS)


def run_discovery(work_dir: str, goal_raw: str) -> Optional[DiscoveryReport]:
    """Run the full discovery pipeline. Returns None on specific goals or any failure.

    Digest is always compiled (injected as context even for specific goals).
    Lens panel + synthesis only run when goal is vague/absent.
    """
    try:
        from agents.discovery import digest as digest_mod
        LOG.info("Discovery: compiling digest for %s", work_dir)
        compiled = digest_mod.compile_digest(work_dir)
    except Exception as e:  # noqa: BLE001
        LOG.warning("Discovery digest failed (%s); skipping discovery", e)
        return None

    if not shouldRunFullDiscovery(goal_raw):
        LOG.info("Discovery: specific goal detected — skipping lens panel")
        return None

    try:
        from agents.discovery import lenses as lenses_mod
        from agents.discovery import synthesis as synthesis_mod
        LOG.info("Discovery: running lens panel")
        findings = lenses_mod.run_lens_panel(compiled)
        LOG.info("Discovery: synthesizing %d findings", len(findings))
        report = synthesis_mod.synthesize(findings)
        LOG.info("Discovery: complete — %d findings, theme=%r",
                 len(report.findings), report.theme_suggestion)
        return report
    except Exception as e:  # noqa: BLE001
        LOG.warning("Discovery lens/synthesis failed (%s); run proceeds without report", e)
        return None
```

```python
                "2. Synthesize them into ONE architectural narrative — a single story this PR will tell\n"
                "3. Prioritize transformative findings over incremental ones\n"
                "4. The resulting work should feel like something an ambitious engineer chose to do,\n"
                "   not a list of chores\n\n"
                "Avoid picking unrelated items. A PR that does one bold thing well is better than\n"
                "five safe things loosely grouped."
            )
        except Exception:  # noqa: BLE001
            discovery_block = ""

    user = (
        f"Specification:\n{goal_elaborated}"
        + discovery_block
        + "\n\nWrite the architecture-design prompt."
    )
    if chat is not None:
        resp = chat(_ARCHITECT_PROMPT_SYSTEM, user)
    else:
        resp = _getOllama().chat(
            model=config.OLLAMA_MODEL,
            messages=[{"role": "system", "content": knowledge.injectCapabilities(_ARCHITECT_PROMPT_SYSTEM)},
                      {"role": "user", "content": user}],
            options={"num_ctx": config.OLLAMA_NUM_CTX},
            keep_alive=config.OLLAMA_KEEP_ALIVE,
        )
    prompt = llm_io.responseText(resp).strip()
    if not prompt:
        LOG.warning("architect-prompt brain output empty -> deterministic goal-bearing fallback")
        prompt = (
            "Produce a thorough architecture and design document for the following specification. "
            "Cover the approach, the modules/files to add or change, data flow, key dependencies, "
            "risks, and an ORDERED implementation outline.\n\nSpecification:\n"
            + (goal_elaborated or "").strip()
            + discovery_block
        )
    return prompt
```
