> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Give 9TO5 mid-subtask crash durability, worktree-isolated parallel workers, and a cost-aware JSON span-tree trace by adding a step-level journal, a resume-from-last-step path, git-worktree isolation with a file-disjointness + atomic-CAS claim, and a trajectory recorder — every unit fail-open behind a flag.

**Architecture:** Four small additive units that share the executor stdout tap (`notifications/subprocess_feed.py`) + `state/experience.db` + git. New modules respect the established `experience_<domain>.py` submodule + facade re-export pattern, carry `@AI-MAP v1` headers, and stay under the `≤450-line file / ≤70-line function` thresholds. No new source of truth: the state store still owns run state, the journal/trajectory are durable replay artifacts only, and the resume *decision* derives from committed state + a plan-signature guard. Sequenced low→high risk: (d) trajectory → (a) journal → (b) resume → (c) worktree/parallel-safety.

**Tech Stack:** Python 3, stdlib `sqlite3` (WAL pool already built in `memory/experience_schema.py`), stdlib `subprocess` for `git worktree`, stdlib `json`/`hashlib`/`tempfile`/`os.replace` for atomic span-tree writes, pytest (run via pytest, **never** bare unittest — local `.env` carries live Discord/Ollama creds and the conftest sandbox is pytest-only).

```python
# platform_core/runtime/trajectory.py
"""Eliza-style file-based JSON span-tree trajectory recorder (Epic 1, unit d).

One append-only span tree per run at ``state/trajectories/<run_id>.json``, written with
an atomic tmp+rename so a crash mid-write never corrupts the file. Records a per-step
token/USD cost series with orchestrator->executor spawn linking. EVERY method swallows
its own errors -- the recorder must never crash a run.

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  Recorder(run_id: str, root=None)  -- Open/close spans; atomic-write the span tree.
DEPENDS-ON: config
SIDE-EFFECTS: writes state/trajectories/<run_id>.json
RUN/TEST: python -m pytest tests/test_trajectory.py
"""
from __future__ import annotations

import json
import os
import threading
import uuid
from datetime import datetime
from pathlib import Path
from typing import Any, Optional

import config

LOG = config.getLogger("runtime.trajectory")

SPAN_KINDS = ("architect", "planner", "executor", "tool", "verify",
              "adversary", "replan", "remediation")


def _nowIso() -> str:
    return datetime.now().astimezone().isoformat(timespec="seconds")


class Recorder:
    """Append-only JSON span tree for one run. Thread-safe; every method is fail-open."""

    def __init__(self, run_id: str, root: Optional[Path] = None) -> None:
        self.run_id = run_id
        self._root = Path(root) if root is not None else (config.STATE_DIR / "trajectories")
        self._lock = threading.Lock()
        self._spans: dict[str, dict[str, Any]] = {}
        self._order: list[str] = []

    def _path(self) -> Path:
        return self._root / f"{self.run_id}.json"

    def openSpan(self, kind: str, parent_id: Optional[str], attrs: Optional[dict] = None) -> str:
        """Open a span; returns its step_id (a child of parent_id, or a root span)."""
        step_id = uuid.uuid4().hex
        try:
            span = {
                "step_id": step_id, "parent_id": parent_id,
                "kind": kind if kind in SPAN_KINDS else "tool",
                "attrs": dict(attrs or {}), "opened_at": _nowIso(),
                "closed_at": None, "usage": None, "cost_usd": None,
            }
            with self._lock:
                self._spans[step_id] = span
                self._order.append(step_id)
                self._flush()
        except Exception as e:  # noqa: BLE001 -- recorder must never crash a run
            LOG.warning("trajectory.openSpan failed (ignored): %s", e)
        return step_id

    def closeSpan(self, step_id: str, usage: Optional[dict] = None,
                  cost_usd: Optional[float] = None) -> None:
        """Close a span, attaching its token usage + USD cost (real 0 for local Ollama)."""
        try:
            with self._lock:
                span = self._spans.get(step_id)
                if span is None:
                    return
                span["closed_at"] = _nowIso()
                span["usage"] = usage
                span["cost_usd"] = cost_usd
                self._flush()
        except Exception as e:  # noqa: BLE001
            LOG.warning("trajectory.closeSpan failed (ignored): %s", e)

    def _flush(self) -> None:
        """Atomic tmp+rename whole-file write (caller holds the lock)."""
        self._root.mkdir(parents=True, exist_ok=True)
        payload = {"run_id": self.run_id,
                   "spans": [self._spans[s] for s in self._order]}
        path = self._path()
        tmp = path.with_suffix(".json.tmp")
        tmp.write_text(json.dumps(payload, ensure_ascii=False, default=str), encoding="utf-8")
        os.replace(tmp, path)
```

# platform_core/runtime/cost_budget.py
"""Swarmclaw-style per-run cost budget (Epic 1, unit d bolt-on).

Rolling sum over the trajectory cost series; emits an 80%-warn and a hard-ceiling
Event on the live bus. Config-gated (COST_BUDGET_ENABLED, OFF by default) and fail-open:
a budget alert is advisory and NEVER aborts a run.

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  CostBudget(run_id: str, bus=None)  -- add(cost_usd) accumulates + emits warn/ceiling events.
DEPENDS-ON: config, notifications.event_bus
SIDE-EFFECTS: emits Events on the live bus
RUN/TEST: python -m pytest tests/test_cost_budget.py
"""
from __future__ import annotations

from typing import Any, Optional

import config

LOG = config.getLogger("runtime.cost_budget")

WARN_FRACTION = 0.80


class CostBudget:
    """Accumulate per-run USD; emit one warn + one ceiling event. Fail-open."""

    def __init__(self, run_id: str, bus: Any = None) -> None:
        self.run_id = run_id
        self._bus = bus
        self._total = 0.0
        self._warned = False
        self._ceiling_hit = False

    def add(self, cost_usd: Optional[float]) -> None:
        if not getattr(config, "COST_BUDGET_ENABLED", False):
            return
        try:
            if cost_usd is None:
                return
            self._total += float(cost_usd)
            ceiling = float(getattr(config, "COST_BUDGET_CEILING_USD", 0.0) or 0.0)
            if ceiling <= 0:
                return
            if not self._ceiling_hit and self._total >= ceiling:
                self._ceiling_hit = True
                self._emit("ceiling", f"run cost ${self._total:.2f} hit ceiling ${ceiling:.2f}")
            elif not self._warned and self._total >= ceiling * WARN_FRACTION:
                self._warned = True
                self._emit("warn", f"run cost ${self._total:.2f} >= 80% of ${ceiling:.2f}")
        except Exception as e:  # noqa: BLE001 -- budgeting is advisory, never crashes a run
            LOG.warning("cost_budget.add failed (ignored): %s", e)

    def _emit(self, level: str, text: str) -> None:
        try:
            bus = self._bus
            if bus is None:
                from notifications.event_bus import get_bus
                bus = get_bus()
            from notifications.event_bus import Event
            bus.emit(Event(kind="cost_budget", sid=self.run_id, text=text,
                           data={"level": level, "total_usd": round(self._total, 4)}))
        except Exception as e:  # noqa: BLE001
            LOG.warning("cost_budget emit failed (ignored): %s", e)
```

# memory/experience_journal.py
"""Experience Ledger -- step-level pending-writes journal (Epic 1, unit a).

Durably appends each completed executor step (one Claude tool call) keyed
``(run_id, subtask_id, attempt, step_idx)`` -- LangGraph's `put_writes` shape -- so a
crash-resume upserts idempotently. A replay/observability artifact, NOT run state; every
function is fail-open (a ledger error logs a warning, never raises).

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  appendStep(run_id, subtask_id, attempt, step_idx, kind, tool, summary, input_obj, plan_sig, session_id, db_path=None) -> None
  loadSteps(run_id, subtask_id, attempt, db_path=None) -> list[dict]
  lastStepIdx(run_id, subtask_id, attempt, db_path=None) -> int  -- -1 if none
  markAttemptSuperseded(run_id, subtask_id, attempt, db_path=None) -> None
DEPENDS-ON: config, memory.experience_schema
SIDE-EFFECTS: writes step_journal rows in experience.db
RUN/TEST: python -m pytest tests/test_experience_journal.py
"""
from __future__ import annotations

from pathlib import Path
from typing import Any, Optional

import config
from memory.experience_schema import LOG, _WRITE_LOCK, _connect, _j, _nowIso


def appendStep(run_id: str, subtask_id: str, attempt: int, step_idx: int, kind: str,
               tool: str, summary: str, input_obj: Any, plan_sig: str, session_id: str,
               db_path: Optional[Path] = None) -> None:
    """Idempotently upsert one executor step row. Best-effort: never raises."""
    if not getattr(config, "STEP_JOURNAL_ENABLED", True):
        return
    if not (run_id or "").strip() or not (subtask_id or "").strip():
        return
    try:

def makeSink(run_id: str, subtask_id: str, attempt: int, plan_sig: str,

"""Resume-from-last-step reconcile (Epic 1, unit b).

Replaces the blind in_progress->pending reset with a journal-aware reconcile: a crashed
subtask whose plan signature still matches its journal is kept resumable (the executor
continues the SAME Claude session and is TOLD what is done -- it does not re-run completed
edits, which are already on disk); a changed plan signature safely rerolls the subtask to a
clean attempt. The state store still owns every mutation (via updateSubtask); the journal is
a replay artifact only. Fail-open: any error degrades to today's safe reset.

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  planSignature(state) -> str
  resumePlan(state, subtask, steps) -> dict
  reconcileResume(state, db_path=None) -> None
DEPENDS-ON: config, memory.experience, state.task
SIDE-EFFECTS: mutates subtask session/status via state.updateSubtask
RUN/TEST: python -m pytest tests/test_step_resume.py
"""
from __future__ import annotations

import hashlib
import json
from pathlib import Path
from typing import Any, Optional

import config
from memory import experience

LOG = config.getLogger("orchestrator.step_resume")

_FOOTER = ("Continue from the current step. Do NOT rewrite the plan or redo completed steps.")
_RESUME_BLOCK_MAX = 4000   # token-budget guard (chars)


def planSignature(state) -> str:
    """Stable hash over the task graph: each subtask's id, title, depends_on, verify.cmd.
    Recorded on every journal row at write time; compared on resume (MAF graph-sig gate)."""
    try:
        graph = []
        for s in state.subtasks:
            graph.append([
                s.get("subtask_id"), s.get("title"),
                sorted(s.get("depends_on") or []),
                (s.get("verify") or {}).get("cmd"),
            ])
        blob = json.dumps(graph, sort_keys=True, ensure_ascii=False, default=str)
        return hashlib.sha256(blob.encode("utf-8")).hexdigest()
    except Exception as e:  # noqa: BLE001 -- a bad sig must degrade to a clean rerun, never crash
        LOG.warning("planSignature failed (ignored): %s", e)
        return ""


def resumePlan(state, subtask: dict, steps: list[dict]) -> dict:
    """AGX-style reconstruction: a token-budgeted [x]/[->]/[ ] block + stability footer.
    Returns {resume_block, session_id, current_idx}."""
    done_idx = [int(s.get("step_idx", -1)) for s in steps if s.get("step_idx") is not None]
    current_idx = (max(done_idx) + 1) if done_idx else 0
    lines = ["## Resume context (you crashed mid-subtask; continue -- do NOT redo done steps)"]
    for s in steps:
        lines.append(f"- [x] step {s.get('step_idx')}: {s.get('tool')} -- {s.get('summary')}")
    lines.append(f"- [→] step {current_idx}: (resume here)")
    lines.append(f"- [ ] remaining steps as needed")
    lines.append("")
    lines.append(_FOOTER)
    block = "\n".join(lines)[:_RESUME_BLOCK_MAX]
    session_id = (steps[-1].get("session_id") if steps else None) \
        or (subtask.get("session") or {}).get("resume_session_id")
    return {"resume_block": block, "session_id": session_id, "current_idx": current_idx}


def reconcileResume(state, db_path: Optional[Path] = None) -> None:
    """For each in_progress subtask: keep it resumable on a plan-sig match (stash resume_block +
    session id on subtask['session'], status->pending), else reset clean + supersede the journal."""
    sig = planSignature(state)
    for s in list(state.subtasks):
        if s.get("status") != "in_progress":
            continue
        sid = s.get("subtask_id")
        try:
            steps = experience.loadSteps(state.task_id, sid, 1, db_path=db_path) \
                if db_path is not None else experience.loadSteps(state.task_id, sid, 1)
        except Exception:  # noqa: BLE001
            steps = []
        row_sig = steps[-1].get("plan_sig") if steps else None
        matches = bool(steps) and sig != "" and row_sig == sig
        sess = dict(s.get("session") or {})
        if matches:
            plan = resumePlan(state, s, steps)
            sess.update({"resumable": True, "resume_block": plan["resume_block"],
                         "resume_session_id": plan["session_id"],
                         "resume_from_idx": plan["current_idx"]})
            state.updateSubtask(sid, status="pending", session=sess)
            LOG.info("resume: %s kept resumable from step %d", sid, plan["current_idx"])
        else:
            sess.update({"resumable": False, "resume_block": None, "resume_from_idx": None})
            state.updateSubtask(sid, status="pending", session=sess)
            try:
                if db_path is not None:
                    experience.markAttemptSuperseded(state.task_id, sid, 1, db_path=db_path)
                else:
                    experience.markAttemptSuperseded(state.task_id, sid, 1)
            except Exception:  # noqa: BLE001
                pass
            LOG.info("resume: %s reset clean (sig mismatch / empty journal)", sid)
    return None
```

> Design note (deliberate, keeps the scheduler unchanged): the spec's conceptual `resumable` status is realized as `session["resumable"] = True` while the subtask status returns to `pending`, so `state.nextActionable()`/`_readySet` (which only schedule `pending`) pick it up unmodified. The resume *context* rides on `subtask["session"]`; the state store still owns the mutation.

# platform_core/orchestration/parallel_safety.py
"""File-disjointness parallel-safety + atomic CAS claim (Epic 1, unit c; guild port).

Two subtasks run concurrently only if their planner-authored file-sets are disjoint AND
neither depends on the other. LOCKED DECISION: file-sets are planner-authored via the
`files_touched` subtask field; an EMPTY/unknown set is treated as conflicting, so an
un-annotated plan conservatively SERIALIZES (never less safe than today). claim() is the
race-free pending->in_progress flip; the caller holds self._state_lock so the read-modify
-write is atomic. State-store ownership is preserved (mutation via state.updateSubtask).

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  filesTouched(subtask) -> set[str]
  disjoint(a, b) -> bool
  safeParallelSet(state, ready, limit) -> list[dict]
  claim(state, sid) -> bool
DEPENDS-ON: config, state.task
SIDE-EFFECTS: claim() flips subtask status via state.updateSubtask
RUN/TEST: python -m pytest tests/test_parallel_safety.py
"""
from __future__ import annotations

from typing import Any

import config

LOG = config.getLogger("orchestrator.parallel_safety")


def filesTouched(subtask: dict) -> set[str]:
    """Planner-authored file-set; empty/missing -> empty set (treated as conflicting)."""
    files = subtask.get("files_touched") or []
    if not isinstance(files, list):
        return set()
    return {str(f) for f in files if f}


def disjoint(a: dict, b: dict) -> bool:
    """True only if BOTH file-sets are known and share no file. Unknown set -> conflicting."""
    fa, fb = filesTouched(a), filesTouched(b)
    if not fa or not fb:
        return False
    return fa.isdisjoint(fb)


def _depConflict(a: dict, b: dict) -> bool:
    """True if either subtask depends on the other (cannot co-run)."""
    return (a.get("subtask_id") in (b.get("depends_on") or [])
            or b.get("subtask_id") in (a.get("depends_on") or []))


def safeParallelSet(state, ready: list[dict], limit: int) -> list[dict]:
    """Greedily pick ready subtasks whose file-sets are pairwise disjoint + dependency-free.
    Always returns at least the first ready subtask (serialize on conflict). Fail-open."""
    if not ready:
        return []
    try:
        picked: list[dict] = [ready[0]]
        for cand in ready[1:]:
            if len(picked) >= max(1, limit):
                break
            if all(disjoint(cand, p) and not _depConflict(cand, p) for p in picked):
                picked.append(cand)
        return picked
    except Exception as e:  # noqa: BLE001 -- on any error, serialize to today's safe single pick
        LOG.warning("safeParallelSet failed (%s); serializing", e)
        return [ready[0]]


def claim(state, sid: str) -> bool:
    """Atomic CAS: flip pending->in_progress ONLY if still pending (else lost-race=False).
    Caller MUST hold self._state_lock so the re-read + write is atomic (rowcount-0 analogue)."""
    st = state.getSubtask(sid)
    if st is None or st.get("status") != "pending":
        return False
    state.updateSubtask(sid, status="in_progress", claimed_by="worker")
    return True
```

# platform_core/runtime/worktree.py
"""Git-worktree-per-subtask isolation backend (Epic 1, unit c; hermes/guild pattern port).

Each parallel worker runs in its own git worktree off HEAD under
``state/.sc-worktrees/<task8>/<sid>``; a successful subtask merges its branch back, a failed
one is left for inspection. Modeled on hermes' BaseEnvironment interface so a container backend
can be added later (Epic 6). Fail-open: ANY git error degrades to the shared work_dir (today's
behavior). gc/remove refuse any path outside state/.sc-worktrees/ (path-scoped safe-remove).

@AI-MAP v1  (generated by tools/repo_map.py -- edit SIDE-EFFECTS only, then regenerate)
EXPORTS:
  enterWorktree(state, sid) -> str  -- isolated cwd (or the shared work_dir on fail-open).
  exitWorktree(path, *, merge) -> None
  removeWorktree(path) -> None
  gcOrphans(state) -> None
  isIsolated(path) -> bool
DEPENDS-ON: config
SIDE-EFFECTS: creates/removes git worktrees under state/.sc-worktrees/
RUN/TEST: python -m pytest tests/test_worktree.py
"""
from __future__ import annotations

import shutil
import subprocess
from pathlib import Path

import config

LOG = config.getLogger("runtime.worktree")


def _worktreesRoot() -> Path:
    return Path(config.STATE_DIR) / ".sc-worktrees"


def _git(args: list[str], cwd: Path) -> subprocess.CompletedProcess:
    return subprocess.run(["git"] + args, cwd=str(cwd), check=True,
                          capture_output=True, text=True)


def _branchFor(task8: str, sid: str) -> str:
    return f"wt/{task8}/{sid}"


def _sharedDir(state) -> str:
    return state.meta.get("work_dir") or str(config.PROJECT_ROOT)


def enterWorktree(state, sid: str) -> str:
    """`git worktree add` a branch off HEAD into state/.sc-worktrees/<task8>/<sid>.
    Returns the worktree path, or the shared work_dir on any git error (fail-open)."""
    try:
        task8 = str(state.task_id)[:8]
        dest = _worktreesRoot() / task8 / sid
        dest.parent.mkdir(parents=True, exist_ok=True)
        if dest.exists():   # stale leftover from a prior crash -> reuse path, recreate clean
            removeWorktree(str(dest))
        branch = _branchFor(task8, sid)
        repo = Path(config.PROJECT_ROOT)
        _git(["worktree", "add", "-b", branch, str(dest), "HEAD"], repo)
        LOG.info("worktree: %s -> %s", sid, dest)
        return str(dest)
    except Exception as e:  # noqa: BLE001 -- degrade to the shared tree (today's behavior)
        LOG.warning("enterWorktree(%s) failed (%s); using shared tree", sid, e)
        return _sharedDir(state)


def exitWorktree(path: str, *, merge: bool) -> None:
    """On success merge the worktree branch back into the run branch, then remove the worktree.
    On failure leave it for inspection. Fail-open; never raises."""
    if not isIsolated(path):
        return
    try:
        repo = Path(config.PROJECT_ROOT)
        p = Path(path)
        if merge:
            # Branch name == wt/<task8>/<sid> derived from the worktree dir layout.
            sid = p.name
            task8 = p.parent.name
            branch = _branchFor(task8, sid)
            _git(["merge", "--no-edit", branch], repo)
        _git(["worktree", "remove", "--force", str(p)], repo)
    except Exception as e:  # noqa: BLE001
        LOG.warning("exitWorktree(%s, merge=%s) failed (%s); leaving for inspection", path, merge, e)


def removeWorktree(path: str) -> None:
    """Path-scoped safe-remove: refuse any path outside state/.sc-worktrees/. Never raises."""
    if not isIsolated(path):
        LOG.warning("removeWorktree refused non-worktree path: %s", path)
        return
    try:
        repo = Path(config.PROJECT_ROOT)
        try:
            _git(["worktree", "remove", "--force", path], repo)
        except Exception:  # noqa: BLE001 -- not a registered worktree -> just rmtree the dir
            shutil.rmtree(path, ignore_errors=True)
    except Exception as e:  # noqa: BLE001
        LOG.warning("removeWorktree(%s) failed (ignored): %s", path, e)


def gcOrphans(state) -> None:
    """Remove leftover worktrees for THIS task (TTL/whitelist cleanup). Never raises."""
    try:
        task8 = str(state.task_id)[:8]
        base = _worktreesRoot() / task8
        if base.exists():
            for child in base.iterdir():
                removeWorktree(str(child))
    except Exception as e:  # noqa: BLE001
        LOG.warning("gcOrphans failed (ignored): %s", e)


def isIsolated(path: str) -> bool:
    """True only when ``path`` lives under state/.sc-worktrees/ (an isolated worktree)."""
    try:
        Path(path).resolve().relative_to(_worktreesRoot().resolve())
        return True
    except Exception:  # noqa: BLE001 -- ValueError (not a subpath) or a resolve error
        return False
```
